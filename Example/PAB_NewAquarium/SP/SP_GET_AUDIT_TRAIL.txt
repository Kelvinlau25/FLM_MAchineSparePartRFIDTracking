CREATE OR REPLACE PROCEDURE SP_GET_AUDIT_TRAIL(
    TABLE_NAME_VAR IN NVARCHAR2,
    KEY_VALUE IN NUMBER DEFAULT 0,
    SORT_COLUMN IN NVARCHAR2,
    SORT_TYPE IN NVARCHAR2,
    SREF OUT SYS_REFCURSOR
)
IS
    SQL_STMT VARCHAR2(32767);
    COUNT_TABLE NUMBER;
    WHERE_CLAUSE NVARCHAR2(32767) := '';
BEGIN
    
    SELECT COUNT(*)
    INTO COUNT_TABLE
    FROM ALL_TABLES
    WHERE TABLE_NAME = TABLE_NAME_VAR;
    
    IF KEY_VALUE <> 0 THEN
        DBMS_OUTPUT.PUT_LINE('is here COUNT ' || COUNT_TABLE);
                DBMS_OUTPUT.PUT_LINE('is here COUNT ' || TABLE_NAME_VAR);
        WHERE_CLAUSE := 'AND KEY_VALUE = ' || KEY_VALUE;
    END IF;
    
    IF COUNT_TABLE > 0 THEN
        SQL_STMT := 'SELECT * FROM ' || TABLE_NAME_VAR || ' WHERE 1 = 1 ' || WHERE_CLAUSE || 
                    ' ORDER BY ' || SORT_COLUMN || ' ' || SORT_TYPE;
        DBMS_OUTPUT.PUT_LINE(SQL_STMT);
        
        OPEN SREF FOR SQL_STMT;
    ELSE
        DBMS_OUTPUT.PUT_LINE('STATUS: 0');
    END IF;
EXCEPTION
    WHEN OTHERS THEN
        DBMS_OUTPUT.PUT_LINE('An error occurred: ' || SQLERRM);
END SP_GET_AUDIT_TRAIL;
